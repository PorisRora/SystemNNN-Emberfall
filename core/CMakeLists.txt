cmake_minimum_required(VERSION 3.16)
project(nnn_script_core VERSION 0.1.0 LANGUAGES C CXX)

include(CTest)
option(NNN_ENABLE_SANITIZERS "Instrument the headless cores and tests with ASan/UBSan" OFF)

add_library(nnn-script-core STATIC src/loader.cpp src/interpreter.cpp src/function_registry.cpp src/scheduler.cpp src/music.cpp src/voice.cpp src/voice_profile.cpp src/resource_path.cpp)
target_include_directories(nnn-script-core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_features(nnn-script-core PUBLIC cxx_std_17)
if(UNIX)
    target_link_libraries(nnn-script-core PUBLIC m)
endif()
set_target_properties(nnn-script-core PROPERTIES CXX_EXTENSIONS OFF)

    foreach(nnn_dependency IN ITEMS zlibstatic png_static)
        target_compile_options(${nnn_dependency} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${nnn_dependency} INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
