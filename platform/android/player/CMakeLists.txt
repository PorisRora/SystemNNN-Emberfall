cmake_minimum_required(VERSION 3.16)
project(systemnnn_player LANGUAGES C CXX)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(MOVIE_CODEC_LINKAGE "shared" CACHE STRING "FFmpeg library linkage (shared or static)")
set_property(CACHE MOVIE_CODEC_LINKAGE PROPERTY STRINGS shared static)
if(MOVIE_CODEC_LINKAGE STREQUAL "shared")
    set(movie_codec_suffix ".so")
elseif(MOVIE_CODEC_LINKAGE STREQUAL "static")
    set(movie_codec_suffix ".a")
else()
    message(FATAL_ERROR "MOVIE_CODEC_LINKAGE must be shared or static")
endif()

add_subdirectory(../../scene scene)
add_library(systemnnn SHARED player_jni.cpp ui_preview_jni.cpp movie_jni.cpp)
target_compile_features(systemnnn PRIVATE cxx_std_17)
target_include_directories(systemnnn PRIVATE "${MOVIE_CODEC_ROOT}/include")
set(movie_codec_libraries)
foreach(component avformat avcodec swscale swresample avutil)
    list(APPEND movie_codec_libraries "${MOVIE_CODEC_ROOT}/lib/lib${component}${movie_codec_suffix}")
endforeach()
if(MOVIE_CODEC_LINKAGE STREQUAL "static")
    # The FFmpeg archives refer to each other and depend on the NDK system libraries.
    target_link_libraries(systemnnn PRIVATE "-Wl,--start-group" ${movie_codec_libraries} "-Wl,--end-group"
                          m atomic mediandk)
else()
    target_link_libraries(systemnnn PRIVATE ${movie_codec_libraries})
endif()
target_link_libraries(systemnnn PRIVATE emberfall-scene android jnigraphics log)
target_link_options(systemnnn PRIVATE -Wl,--no-undefined -Wl,--exclude-libs,ALL -Wl,-z,max-page-size=16384)
