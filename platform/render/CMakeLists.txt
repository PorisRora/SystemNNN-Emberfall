cmake_minimum_required(VERSION 3.16)
project(emberfall_render LANGUAGES CXX)
include(CTest)
option(EMBERFALL_RENDER_SANITIZERS "Instrument compositor and tests" OFF)
add_library(emberfall-render STATIC src/render.cpp src/layers.cpp src/effects.cpp src/wipes.cpp)
target_include_directories(emberfall-render PUBLIC include PRIVATE ../../core/include)
target_compile_features(emberfall-render PUBLIC cxx_std_17)
set_target_properties(emberfall-render PROPERTIES CXX_EXTENSIONS OFF POSITION_INDEPENDENT_CODE ON)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(emberfall-render PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Werror)
endif()
if(EMBERFALL_RENDER_SANITIZERS)
    target_compile_options(emberfall-render PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(emberfall-render PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
endif()
if(BUILD_TESTING)
    add_executable(emberfall-render-tests tests/render_tests.cpp)
    target_link_libraries(emberfall-render-tests PRIVATE emberfall-render)
    add_test(NAME emberfall-render-behavior COMMAND emberfall-render-tests)
    set_tests_properties(emberfall-render-behavior PROPERTIES TIMEOUT 30)
endif()
